cmake_minimum_required(VERSION 3.24)
project(CodexMonitorNative VERSION 2.0.0 LANGUAGES CXX RC)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "Build with MSVC on Windows")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(CodexMonitor WIN32 src/main.cpp src/core.cpp src/platform.cpp src/render.cpp src/application.cpp resources/app.rc)
target_compile_definitions(CodexMonitor PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
target_compile_options(CodexMonitor PRIVATE /utf-8 /W4 /EHsc /permissive-)
target_link_options(CodexMonitor PRIVATE /MANIFEST:NO)
target_link_libraries(CodexMonitor PRIVATE d2d1 dwrite dwmapi user32 gdi32 shell32 shlwapi ole32 comctl32 comdlg32 advapi32 bcrypt uxtheme psapi iphlpapi ws2_32)
target_include_directories(CodexMonitor PRIVATE vendor src)
set_property(DIRECTORY PROPERTY VS_STARTUP_PROJECT CodexMonitor)
option(CQM_BUILD_TESTS "Build native core regression tests" OFF)
if(CQM_BUILD_TESTS)
  enable_testing()
  add_executable(CodexMonitorCoreTests tests/core_tests.cpp src/core.cpp)
  target_compile_definitions(CodexMonitorCoreTests PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_compile_options(CodexMonitorCoreTests PRIVATE /utf-8 /W4 /EHsc)
  target_include_directories(CodexMonitorCoreTests PRIVATE vendor src)
  target_link_libraries(CodexMonitorCoreTests PRIVATE bcrypt)
  add_test(NAME native_core COMMAND CodexMonitorCoreTests)
endif()
